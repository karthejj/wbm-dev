SELECT mat.material_name MaterialName, 
count(*) as No_Transaction,
sum(cht.material_weight) as Qty,
sum(cht.material_price_after_vat) as MaterialPrice
FROM postgreswbm.child_transaction cht,
postgreswbm.material mat,
postgreswbm.transactions_header hd
where mat.id = cht.material_id and cht.transaction_header_id = hd.id and hd.transfer_type = 'OUT' and mat.id<>16
group by mat.material_name

SELECT mat.material_name MaterialName, 
count(*) as No_Transaction,
sum(cht.material_weight) as Qty,
sum(cht.material_price_after_vat) as MaterialPrice
FROM postgreswbm.child_transaction cht,
postgreswbm.material mat,
postgreswbm.transactions_header hd
where mat.id = cht.material_id and cht.transaction_header_id = hd.id 
and hd.transfer_type = 'WEIGH'
and mat.id=16
group by mat.material_name 



select mat.material_name, (case when (listOne.qty1-listTwo.qty2) is null then listOne.qty1
				   			when (listOne.qty1-listTwo.qty2) is not null then listOne.qty1-listTwo.qty2
					END) as availableInYard
from (
	SELECT mat.id as id,
sum(cht.material_weight) as qty1
FROM postgreswbm.child_transaction cht,
postgreswbm.material mat,
postgreswbm.transactions_header hd
where mat.id = cht.material_id 
and cht.transaction_header_id = hd.id and hd.transfer_type = 'INC'
group by mat.id
) listOne left join
(
	SELECT mat.id as id,
sum(cht.material_weight) as qty2
FROM postgreswbm.material mat 
left join postgreswbm.child_transaction cht on cht.material_id = mat.id 
join postgreswbm.transactions_header hd on (hd.id = cht.transaction_header_id 
and hd.transfer_type = 'OUT')
group by mat.id
) listTwo
on listOne.id = listTwo.id
join postgreswbm.material mat on mat.id = listOne.id and mat.id<>16

get the distinct user names from the child transaction

select distinct uname from (
select
distinct cht.created_by as uname
FROM postgreswbm.child_transaction cht
where cht.created_by is not null
union
select
distinct cht.modified_by as uname
FROM postgreswbm.child_transaction cht
where cht.created_by is not null
) abc where uname is not null
order by uname
